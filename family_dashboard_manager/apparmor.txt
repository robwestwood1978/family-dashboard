  /run/{s6,s6-rc*,service}/** rix,
  /package/** rix,
  /command/** rix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,

  /usr/lib/** mr,
  /lib/** mr,
  /lib64/** mr,
  /dev/null rw,
  /dev/tty rw,
  /dev/urandom r,
  /etc/{,**} r,
  /usr/lib/bashio/** rix,
  /opt/family-dashboard/{,**} rix,
  /tmp/{,**} rwk,
  /data/{,**} rwk,

  /config/family-dashboard/ rw,
  /config/family-dashboard/** rwk,
}
